class Solution:
    def isMatch(self, s: str, p: str) -> bool:
        #space optimized dp solution
        m, n = len(s), len(p)
        prev = [False] * (n + 1)
        prev[n] = True

        for i in range(m, -1, -1):
            curr = [False] * (n + 1)
            if i == m:
                curr[n] = True

            for j in range(n - 1, -1, - 1):
                first = i < m and p [j] in {s[i], '.'}
                if j + 1 < n and p[j + 1] == '*':
                    curr[j] = curr[j + 2] or (first and prev[j])
                else:
                    curr[j] = first and prev[j + 1]

            prev = curr

        return prev[0]
