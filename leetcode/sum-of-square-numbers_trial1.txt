class Solution:
    def judgeSquareSum(self, c: int) -> bool:
        a = 0
        b = int(math.sqrt(c))
        
        while a <= b:
            sum_squares = a*a + b*b
            if sum_squares == c:
                return True
            elif sum_squares < c:
                a += 1
            else:
                b -= 1
        
        return False
