class Solution:
    def maxArea(self, height: List[int]) -> int:
        max_amount = 0
        l = 0
        r = len(height) - 1
        while l < r:
            capacity = (r-l)*(min(height[l], height[r]))
            max_amount = max(capacity, max_amount)
        
            if height[l] < height[r]:
                l += 1
            else:
                r -= 1
        return max_amount
