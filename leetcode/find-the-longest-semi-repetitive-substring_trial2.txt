class Solution:
    def longestSemiRepetitiveSubstring(self, s: str) -> int:
        n = len(s)
        if n <= 1:
            return n

        start = 0
        count_pairs = 0
        max_len = 1
        
        for end in range(1, n):
            if s[end] == s[end - 1]:
                count_pairs += 1   
            while count_pairs > 1:
                if s[start] == s[start + 1]:
                    count_pairs -= 1
                start += 1 
            max_len = max(max_len, end - start + 1) 
        return max_len
