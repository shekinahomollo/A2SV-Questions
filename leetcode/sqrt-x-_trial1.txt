class Solution:
    def mySqrt(self, x: int) -> int:
        if x == 0:
            return 0

        l = 1
        r = x
        while l <= r:
            m = (l + r) // 2
            if x // m == m:
                return m
            elif x // m > m:
                l = m + 1
            else:
                r = m - 1

        return r
