class Solution:
    def maximumUniqueSubarray(self, nums: List[int]) -> int:
        seen = set()
        l = 0
        res = 0
        max_score = 0

        for r in range(len(nums)):
            while nums[r] in seen:
                seen.remove(nums[l])
                res -= nums[l]
                l += 1

            seen.add(nums[r])
            res += nums[r]
            max_score = max(max_score, res)

        return max_score
