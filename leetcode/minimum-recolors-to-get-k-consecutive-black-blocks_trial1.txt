class Solution(object):
    def minimumRecolors(self, blocks, k):
        white_count = blocks[:k].count('W')
        min_white = white_count

        for i in range(k, len(blocks)):
            if blocks[i] == 'W':
                white_count += 1 
            if blocks[i - k] == 'W':
                white_count -= 1 
            min_white = min(min_white, white_count) 
        return min_white
