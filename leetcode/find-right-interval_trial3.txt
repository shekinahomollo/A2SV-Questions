class Solution:
    def findRightInterval(self, intervals: List[List[int]]) -> List[int]:
        n = len(intervals)

        #store index and start
        starts = sorted((intervals[i][0], i) for i in range(n))

        #extract start values for binary search
        start_vals = [s[0] for s in starts] 

        res = [-1] * n

        #process each interval
        for i in range(n):
            end = intervals[i][1]

            idx = bisect_left(start_vals, end)
            if idx < n:
                res[i] = starts[idx][1]

        return res
