class Solution:
    def maxTurbulenceSize(self, arr: List[int]) -> int:
        n = len(arr)
        l = 0
        max_len = 1

        def sign(x):
            if x > 0: return 1
            if x < 0: return -1
            return 0

        for r in range(1, n):
            cmp = sign(arr[r] - arr[r - 1])

            if cmp == 0:
                l = r
            elif r == n - 1 or cmp * sign(arr[r + 1] - arr[r]) != -1:
                max_len = max(max_len, r - l + 1)
                l = r

        return max_len
