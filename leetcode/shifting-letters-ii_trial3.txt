class Solution:
    def shiftingLetters(self, s: str, shifts: List[List[int]]) -> str:
        n = len(s)
        alphabet = [0] * n
        result = []
        for l, r, t in shifts:
            if t == 1:
                alphabet[l] += 1 #move it forward
                if r + 1 < n:
                    alphabet[r + 1] -= 1
            else:
                alphabet[l] -= 1
                if r + 1 < n:
                    alphabet[r + 1] += 1
        
        for i in range(1, n):
            alphabet[i] += alphabet[i - 1]

        for i in range(n):
            shift = (alphabet[i] % 26 + 26) % 26 
            ch = (ord(s[i]) - ord('a') + shift) % 26
            result.append(chr(ord('a') + ch))
        return ''.join(result)
