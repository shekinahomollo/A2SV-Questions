class Solution:
    def smallestDivisor(self, nums: List[int], threshold: int) -> int:

        def do_sum(d):
            total = 0
            for num in nums:
                total += (num + d - 1) // d
            return total

        l, r = 1, max(nums)
        ans = r

        while l <= r:
            mid = (l + r) // 2

            if do_sum(mid) <= threshold:
                ans = mid
                r = mid - 1 #try smaller divisor
            else:
                l = mid + 1 #increase divisor

        return ans
