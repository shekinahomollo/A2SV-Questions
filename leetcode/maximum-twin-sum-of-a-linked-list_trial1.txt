# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def pairSum(self, head: Optional[ListNode]) -> int:
        values = []
        curr = head
        while curr:
            values.append(curr.val)
            curr = curr.next

        res = 0
        sum_vals = 0
        i, j = 0, len(values) - 1
        while i <= j:
            sum_vals = values[i] + values[j]
            res = max(res, sum_vals)
            i += 1
            j -= 1
        return res
