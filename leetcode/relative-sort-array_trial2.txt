class Solution:
    def relativeSortArray(self, arr1: List[int], arr2: List[int]) -> List[int]:
        res = []
        counts = [0] * 1001

        for num in arr1:
            counts[num] += 1

        for num in arr2:
            while counts[num] > 0:
                res.append(num)
                counts[num] -= 1

        for i in range(len(counts)):
            while counts[i] > 0:
                res.append(i)
                counts[i] -= 1

        return res
